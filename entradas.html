function registrarReserva(datos) { const ss = SpreadsheetApp.openById(ID_PLANILLA); const hoja = ss.getSheetByName('RESERVAS'); if (!hoja) { throw new Error('No existe la hoja RESERVAS'); } const nombre = String(datos.nombre || '').trim(); const telefono = String(datos.telefono || '').trim(); const medioPago = String(datos.medioPago || 'TRANSFERENCIA').trim(); if (!nombre || !telefono) { throw new Error('Faltan datos de la reserva'); } const lock = LockService.getScriptLock(); lock.waitLock(10000); try { hoja.appendRow([ new Date(), nombre, telefono, medioPago, 'PENDIENTE', '' ]); return { ok: true, mensaje: 'Reserva registrada' }; } finally { lock.releaseLock(); } } function doPost(e) { try { const datos = { nombre: e.parameter.nombre || '', telefono: e.parameter.telefono || '', medioPago: e.parameter.medioPago || 'TRANSFERENCIA' }; const resultado = registrarReserva(datos); return ContentService .createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ ok: false, mensaje: error.message })) .setMimeType(ContentService.MimeType.JSON); } } /* ASIGNACIÓN AUTOMÁTICA DE ENTRADAS Cuando en RESERVAS cambiamos ESTADO a PAGADO: - busca la primera entrada libre entre 1 y 200 - coloca comprador y teléfono en ENTRADAS - marca PAGO = PAGADO - escribe el número en ENTRADA ASIGNADA */ function onEdit(e) { if (!e || !e.range) return; const hoja = e.range.getSheet(); // Solo trabaja dentro de RESERVAS if (hoja.getName() !== 'RESERVAS') return; // Columna E = ESTADO if (e.range.getColumn() !== 5) return; // Ignora encabezado if (e.range.getRow() < 2) return; const estado = String(e.range.getValue()) .trim() .toUpperCase(); if (estado !== 'PAGADO') return; const filaReserva = e.range.getRow(); asignarEntradaPagada_(filaReserva); } function asignarEntradaPagada_(filaReserva) { const lock = LockService.getDocumentLock(); lock.waitLock(10000); try { const ss = SpreadsheetApp.openById(ID_PLANILLA); const reservas = ss.getSheetByName('RESERVAS'); const entradas = ss.getSheetByName('ENTRADAS'); if (!reservas || !entradas) { throw new Error('No se encontraron RESERVAS o ENTRADAS'); } // Si ya tiene entrada asignada, no hace nada. const yaAsignada = reservas .getRange(filaReserva, 6) .getValue(); if (yaAsignada !== '') { return; } const nombre = String( reservas.getRange(filaReserva, 2).getValue() ).trim(); const telefono = String( reservas.getRange(filaReserva, 3).getValue() ).trim(); if (!nombre || !telefono) { throw new Error('La reserva no tiene nombre o WhatsApp'); } /* ENTRADAS: filas 2 a 201 = entradas 1 a 200 A = NÚMERO B = CÓDIGO ÚNICO C = COMPRADOR D = TELÉFONO E = PAGO F = ESTADO DE INGRESO G = FECHA/HORA */ const datosEntradas = entradas.getRange(2, 1, 200, 7).getValues(); let filaEntrada = -1; let numeroEntrada = ''; for (let i = 0; i < datosEntradas.length; i++) { const comprador = String( datosEntradas[i][2] || '' ).trim(); const pago = String( datosEntradas[i][4] || '' ).trim().toUpperCase(); /* Consideramos libre una entrada que todavía no tiene comprador y no está marcada PAGADO. */ if (comprador === '' && pago !== 'PAGADO') { filaEntrada = i + 2; numeroEntrada = datosEntradas[i][0]; break; } } if (filaEntrada === -1) { reservas .getRange(filaReserva, 5) .setValue('SIN ENTRADAS'); throw new Error('No quedan entradas disponibles'); } // Completa la entrada seleccionada. entradas .getRange(filaEntrada, 3) .setValue(nombre); entradas .getRange(filaEntrada, 4) .setValue(telefono); entradas .getRange(filaEntrada, 5) .setValue('PAGADO'); // Guarda en RESERVAS qué entrada pertenece al comprador. reservas .getRange(filaReserva, 6) .setValue(numeroEntrada); } finally { lock.releaseLock(); } }
